void BlendSpace2D::addSample(float x, float y, const AnimationSequence& animation) {
    samples.push_back({x, y, animation});

std::vector<std::pair<AnimationSequence, float>> BlendSpace2D::getBlendedAnimations(float inputX, float inputY) {
    assert(samples.size() >= 4 && "At least 4 samples are required for blending");

    // Find the four nearest samples around the input X, Y values
    std::vector<Sample*> nearestSamples(4);
    std::vector<float> distances(4, std::numeric_limits<float>::max());

    for (Sample& sample : samples) {
        float distance = (sample.x - inputX) * (sample.x - inputX) + (sample.y - inputY) * (sample.y - inputY);

        for (size_t i = 0; i < distances.size(); ++i) {
            if (distance < distances[i]) {
                std::rotate(nearestSamples.begin() + i, nearestSamples.end() - 1, nearestSamples.end());
                std::rotate(distances.begin() + i, distances.end() - 1, distances.end());
                nearestSamples[i] = &sample;
                distances[i] = distance;
                break;
            }
        }
    }

    // Calculate the blend weights using the interpolation method
    std::vector<std::pair<AnimationSequence, float>> blendedAnimations;
    for (const Sample* sample : nearestSamples) {
        float weightX = interpolationMethod(inputX, sample->x, nearestSamples[0]->x, nearestSamples[1]->x, nearestSamples[2]->x);
        float weightY = interpolationMethod(inputY, sample->y, nearestSamples[0]->y, nearestSamples[1]->y, nearestSamples[2]->y);
        float blendWeight = weightX * weightY;
        blendedAnimations.push_back({sample->animation, blendWeight});
    }

    return blendedAnimations;
}

void BlendSpace2D::setInterpolationMethod(const InterpolationMethod& method) {
    interpolationMethod = method;
}

const BlendSpace2D::InterpolationMethod& BlendSpace2D::getInterpolationMethod() const {
    return interpolationMethod;
}

float BlendSpace2D::cubicInterpolation(float a, float b, float c, float d, float t) const {
    float A = (d - c) - (a - b);
    float B = (a - b) - A;
    float C = c - a;
    float D = b;

    return A * t * t * t + B * t * t + C * t + D;
}

Give me a very simple std::rotate example to help explain the concept and its usefulness 
std::rotate is a function in the C++ Standard Library that rotates elements within a specified range. It takes the elements from the beginning of the range to a middle point and moves them to the end of the range, while the elements from the middle to the end of the range move to the beginning. This operation maintains the relative order of elements within the rotated parts.

Here's a simple example to illustrate the concept and usefulness of std::rotate:

#include <vector>
#include <algorithm>

int main() {
    std::vector<int> numbers = {1, 2, 3, 4, 5};

    std::cout << "Original vector: ";
    for (int num : numbers) {
        std::cout << num << " ";
    }
    std::cout << std::endl;

    // Rotate the vector so that the number 3 becomes the first element
    std::rotate(numbers.begin(), numbers.begin() + 2, numbers.end());

    std::cout << "Rotated vector: ";
    for (int num : numbers) {
        std::cout << num << " ";
    }
    std::cout << std::endl;

    return 0;
}
